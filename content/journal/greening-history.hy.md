date = "2025-12-21"
+++
2004 թվականին հրատարակված փաստաթուղթը պատրաստվել է Երևանի պետական համալսարանի և Հայաստանի ագրարային ակադեմիայի աշխատակիցների կողմից։ Այն նկարագրում է Երևանի և նրա շրջակայքի համակարգված կանաչապատման պատմությունը՝ սկսած 1938 թվականից մինչև 1993 թվականը։
