date = "2025-07-17"
+++
Մենք հրատարակել ենք մեր առաջին զեկույցը [«Փողոցի անձնագիր»](/hy/monitor/passports/) նախագծի շրջանակներում։ Այս անգամ մենք կենտրոնացել ենք Երևանի Թումանյան փողոցի վրա։ Զեկույցը ներառում է փողոցի ներկա վիճակի մանրամասն վերլուծություն, ներառյալ նրա կանաչապատումը, ենթակառուցվածքները և բարելավման հնարավոր ուղղությունները։
